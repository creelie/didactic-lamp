    \draw[orange!70!black,blk] (27-\v,-\h) rectangle (27-\u,\h);
    \draw[-{Stealth[length=2mm]},blue!55!black,dashed,line width=0.6pt]
      ({(\u+\v)/2},\h) .. controls ({(\u+\v)/2},{0.9+0.04*(27-\u-\v)}) and ({27-(\u+\v)/2},{0.9+0.04*(27-\u-\v)}) .. ({27-(\u+\v)/2},\h);
